G1 = {"Alice", "Bob", "Charlie", "Diana"}   # outdoor only group

all_games = G1 & G2 & G3

print("All students (union):", all_students)

Paper 3 · Q2b Birthday game — enter DOB, compute single digit sum, display personality message 🔴 Hot 
# Single digit sum: keep adding digits until one digit remains
def single_digit_sum(n):
    while n >= 10:
        n = sum(int(d) for d in str(n))
    return n

dob = input("Enter your date of birth (DDMMYYYY): ")

# Remove non-digits and sum all digits
digits_only = "".join(c for c in dob if c.isdigit())
total = sum(int(d) for d in digits_only)
result = single_digit_sum(total)

# Personality dictionary — maps digit to message
messages = {
    1: "Intelligent", 2: "Intelligent", 3: "Intelligent",
    4: "Cool",        5: "Cool",        6: "Cool",
    7: "Gorgeous",    8: "Gorgeous",    9: "Gorgeous"
}

print(f"Single digit sum: {result}")
print(f"You are: {messages[result]}!")
Enter your date of birth (DDMMYYYY): 15082004 Single digit sum: 2 You are: Intelligent!
💡 single_digit_sum() reduces any number to a single digit by repeatedly summing its digits. e.g. 1+5+0+8+2+0+0+4 = 20 → 2+0 = 2

Paper 7 · Q4b & Paper 8 · Q4b Base: Shape; Subclasses: Rectangle (area = l×w) and Circle (area = πr²) 🔴 Hot 
import math

class Shape:
    def __init__(self, length=0, width=0, radius=0):
        self.length = length
        self.width  = width
        self.radius = radius

    def area(self):
        print("Area not defined for base Shape.")

class Rectangle(Shape):
    def __init__(self, length, width):
        super().__init__(length=length, width=width)

    def area(self):
        result = self.length * self.width
        print(f"Rectangle Area ({self.length} x {self.width}) = {result}")

class Circle(Shape):
    def __init__(self, radius):
        super().__init__(radius=radius)

    def area(self):
        result = round(math.pi * self.radius ** 2, 2)
        print(f"Circle Area (r={self.radius}) = {result}")

r = Rectangle(5, 3)
c = Circle(7)
r.area()
c.area()
Rectangle Area (5 x 3) = 15 Circle Area (r=7) = 153.94

def is_spy(n):
    digits = [int(d) for d in str(n)]
    digit_sum     = sum(digits)
    digit_product = 1
    for d in digits:
        digit_product *= d
    return digit_sum == digit_product

n = int(input("Enter a number: "))
if is_spy(n):
    print(f"{n} is a Spy Number!")
else:
    print(f"{n} is NOT a Spy Number.")
Enter a number: 1124 1+1+2+4 = 8 and 1×1×2×4 = 8 → Equal! 1124 is a Spy Number!
💡 A spy number has sum of digits = product of digits. e.g. 1124: sum=8, product=8 ✓
Paper 2 · Q2b & Paper 8 · Q3a Armstrong number — sum of cubes of digits equals the number 🔴 Hot 
def is_armstrong(n):
    digits = [int(d) for d in str(n)]
    power  = len(digits)          # works for any length, not just 3-digit
    total  = sum(d ** power for d in digits)
    return total == n

n = int(input("Enter a 3-digit number: "))
if is_armstrong(n):
    print(f"{n} is an Armstrong number.")
else:
    print(f"{n} is NOT an Armstrong number.")
Enter a 3-digit number: 153 1³ + 5³ + 3³ = 1 + 125 + 27 = 153 ✓ 153 is an Armstrong number.

def factorial(n):
    if n == 0 or n == 1:    # base case
        return 1
    return n * factorial(n - 1)   # recursive call

n = int(input("Enter a number: "))
print(f"Factorial of {n} = {factorial(n)}")
Enter a number: 5 Factorial of 5 = 120
💡 Trace: factorial(5) → 5×factorial(4) → 5×4×factorial(3) → ... → 5×4×3×2×1 = 120

def reverse(n, rev=0):
    if n == 0:
        return rev          # base case
    digit = n % 10
    return reverse(n // 10, rev * 10 + digit)

n = int(input("Enter a number: "))
print(f"Reverse of {n} = {reverse(n)}")

Paper 6 · Q3a Vehicle registration — validate format AABBCCDDDD (2 letters, 2 digits, 2 letters, 4 digits) 🔴 Hot 
def validate_reg(reg):
    reg = reg.replace(" ", "").upper()   # clean input

    if len(reg) != 10:
        return "Invalid: must be exactly 10 characters"

    # Format: AA BB CC DDDD  →  [0:2]=alpha, [2:4]=digit, [4:6]=alpha, [6:10]=digit
    if   reg[0:2].isalpha()  \
     and reg[2:4].isdigit()  \
     and reg[4:6].isalpha()  \
     and reg[6:10].isdigit():
        return f"'{reg}' is a VALID registration number."
    else:
        return f"'{reg}' is INVALID."

reg = input("Enter vehicle registration number: ")
print(validate_reg(reg))
Enter vehicle registration number: MH12AB1234 'MH12AB1234' is a VALID registration number. Enter vehicle registration number: MH1AB1234 'MH1AB1234' is INVALID.

Paper 4 · Q1a Multiply and divide by 4 WITHOUT arithmetic operators (use bit shifting) 🔴 Hot 
# Bit shift: << shifts left (multiply by 2), >> shifts right (divide by 2)
# Multiply by 4 = shift left by 2 (4 = 2²)
# Divide by 4   = shift right by 2

n = int(input("Enter a number: "))

multiplied = n << 2    # n * 4 using left bit shift
divided    = n >> 2    # n // 4 using right bit shift

print(f"{n} multiplied by 4 = {multiplied}")
print(f"{n} divided by 4   = {divided}")
Enter a number: 20 20 multiplied by 4 = 80 20 divided by 4 = 5
💡 n << k = n × 2^k | n >> k = n ÷ 2^k (integer division). So <<2 = ×4, >>2 = ÷4. No +, -, *, / used.
